cmake_minimum_required(VERSION 3.22.1)
project(lmnop_omi_med LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PARAKEET_SHARED OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "Target CPU instructions")
add_subdirectory("${OMI_ENGINE_SOURCE}" engine)
# Upstream's include path assumes it is the top-level CMake project.
target_include_directories(parakeet PRIVATE "${OMI_ENGINE_SOURCE}/third_party")
add_library(lmnop_omi_med SHARED omi_med_jni.cpp)
if(OMI_LIBRARY_NAME)
    set_target_properties(lmnop_omi_med PROPERTIES OUTPUT_NAME "${OMI_LIBRARY_NAME}")
endif()
add_library(lmnop_omi_med_cpu SHARED cpu_features.cpp)
target_link_options(lmnop_omi_med_cpu PRIVATE "-Wl,-z,max-page-size=16384")
target_include_directories(lmnop_omi_med PRIVATE "${OMI_ENGINE_SOURCE}/src")
target_compile_options(lmnop_omi_med PRIVATE -O3)
# Keep GGML private: whisper.rn ships a separate version in the same process.
target_link_options(lmnop_omi_med PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic" "-Wl,-z,max-page-size=16384")
target_link_libraries(lmnop_omi_med PRIVATE parakeet android log)
