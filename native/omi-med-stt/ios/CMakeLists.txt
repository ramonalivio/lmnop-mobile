cmake_minimum_required(VERSION 3.22.1)
project(LmnopOmiMedEngine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PARAKEET_SHARED OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PARAKEET_GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OMI_ENGINE_SOURCE}" engine)
target_include_directories(parakeet PRIVATE "${OMI_ENGINE_SOURCE}/third_party")
add_library(${OMI_FRAMEWORK_NAME} SHARED omi_engine.cpp)
target_include_directories(${OMI_FRAMEWORK_NAME} PRIVATE "${OMI_ENGINE_SOURCE}/src")
target_link_libraries(${OMI_FRAMEWORK_NAME} PRIVATE parakeet)
# Export only our C API. GGML must stay private alongside whisper.rn/Moonshine.
target_link_options(${OMI_FRAMEWORK_NAME} PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
set_target_properties(${OMI_FRAMEWORK_NAME} PROPERTIES
    FRAMEWORK TRUE
    MACOSX_FRAMEWORK_IDENTIFIER "com.lmnop.${OMI_FRAMEWORK_NAME}"
    MACOSX_FRAMEWORK_SHORT_VERSION_STRING "1.0"
    MACOSX_FRAMEWORK_BUNDLE_VERSION "1"
    PUBLIC_HEADER "omi_engine.h"
    INSTALL_NAME_DIR "@rpath"
    BUILD_WITH_INSTALL_NAME_DIR TRUE
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
